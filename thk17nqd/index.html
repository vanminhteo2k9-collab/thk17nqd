// Lazy Propagation Segment Tree — O(log N)
void update(int id, int l, int r, int u, int v, long long val) {
    if (v < l || r < u) return;
    if (u <= l && r <= v) {
        st[id].sum += val * (r - l + 1);
        st[id].lazy += val;
        return;
    }
    pushDown(id, l, r);
    int mid = (l + r) >> 1;
    update(id << 1, l, mid, u, v, val);
    update(id << 1 | 1, mid + 1, r, u, v, val);
    st[id].sum = st[id << 1].sum + st[id << 1 | 1].sum;
}
